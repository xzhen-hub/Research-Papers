 /******Variable Definition******************/
 /*********************************************
 stcnty:The state and county FIPS code associated with the headquarters location of the institution.
 stcntybr:The state and county FIPS code associated with the specific branch location.
 stnumbr:The FIPS number of the state in which the branch is physically located.
 
 DEPDOM: Total domestic deposits of the institution as of June 30.
 DEPSUM: Total deposits of the institution as of June 30.
 DEPSUMBR: Branch office deposits as of June 30
 
 brnum:Numerical reference to identify a branch office within one institution
 
 CNTYNUMB: County number that corresponds to the county in which the branch is located.
 
 SIMS_Established Date:The date that the branch location was established.
 SIMS_Latititude:The lattitude of the branch's physical location.
 SIMS_Longitude:The longitude of the branch's physical location
 
 RSSDHCR: The unique number assigned by the Federal Reserve Board (FRB) to the top regulatory bank holding company.
 RSSDID: The unique number assigned by the Federal Reserve Board (FRB) to the institution.
 
 CERT:The certificate number assigned to an institution for deposit insurance.
 BRNUM: Numerical reference to identify a branch office within one institution
 BKMO: Main office / branch identifier; Main office  = 1, branch = 0
 UNINUMBR:The unique number associated with a specific physical location of a branch.  This number remains with this location over time, regardless of ownership.
 
 namefull:Institution name.
 namebr:Branch name.
 namehcr:The name of the headquarters of the top regulatory bank holding company.
 
 **********************************************/

 /*************************new variables generated*********************************/
 /*******
 1. depsum_cnty: branch deposit @ county level
 2. depsum_rssd: branch deposit @ rssd level
 3. TD_rssdstate: rssd deposit @ state level (by stnumbr)
 4. distance: the distance between main office and branches
 5. disRSSD_br: weighted average of distance in miles between a facility and its branches
 6. TD_state: branch deposit  state level
 7. dep_dispersion: deposit dispersion measure
 8. div_index_st: generate geographic diversification at state level
 
 ********/

 /***generate new variables represent latitude and longitude if bkmo=1(main office)**********/

drop _merge

 /*************use latitude and longitude to calculate the distance between main office and branches****/
  clear
 cd "D:\Research\Banking power and IPO\dataneed"
 use 1994_dis_dep
gen long_br = atan(1)/45 * sims_latitude
gen lati_br = atan(1)/45 * sims_longitude
gen long_main = atan(1)/45 * latitude_main
gen lati_main = atan(1)/45 * latitude_main

gen dlon=long_main-long_br
gen dlat=lati_main-lati_br
gen a=(sin(dlat/2))^2 + cos(lati_br) * cos(lati_main) * (sin(dlon/2))^2
gen c= 2 * atan2(sqrt(a),sqrt(1-a))
gen distance=3957 * c
/***use the other method to verify the result of distance, it's correct 
 gen distance2=2*3961*asin(sqrt(sin(dlat/2))^2)+cos(lati_br) * cos(lati_main)*(sin(dlon/2))^2
 drop distance2
 ***/
 codebook distance
 save 1994_dis_dep, replace

 /*************use latitude and longitude to calculate the distance between main office and branches****/
  clear
 cd "D:\Research\Banking power and IPO\dataneed"

gen long_br = atan(1)/45 * sims_latitude
gen lati_br = atan(1)/45 * sims_longitude
gen long_main = atan(1)/45 * latitude_main
gen lati_main = atan(1)/45 * latitude_main

gen dlon=long_main-long_br
gen dlat=lati_main-lati_br

 drop distance2

/**********************************************************************************/
 clear
 cd "D:\Research\Banking power and IPO\dataneed"
 use  all_1999_dep
 

 keep year rssdhcr rssdid cert brnum bkmo uninumbr sims_latitude sims_longitude depsumbr depsum stcnty stcntybr stnumbr namebr namehcr namefull
 save 1999_only, replace

 /***generate new variables represent latitude and longitude if bkmo=1(main office)**********/
sort rssdid year
by rssdid year: gen latitude_main1=sims_latitude if bkmo==1
by rssdid year: gen longitude_main1=sims_longitude if bkmo==1
collapse (sum) latitude_main=latitude_main1 longitude_main=longitude_main1, by (rssdid year)
duplicates report rssdid
save lalomain_rssd_1999, replace


/*********join the all file with lalo file to generate distance******/ 
 clear
 cd "D:\Research\Banking power and IPO\dataneed"
 use all_1999_dep
sort rssdid year
joinby rssdid year using  lalomain_rssd_1999, unm(master)
drop _merge
save 1999_dis_dep, replace
 
 
 /*************use latitude and longitude to calculate the distance between main office and branches****/
  clear
 cd "D:\Research\Banking power and IPO\dataneed"
 use 1999_dis_dep
gen long_br = atan(1)/45 * sims_latitude
gen lati_br = atan(1)/45 * sims_longitude
gen long_main = atan(1)/45 * latitude_main
gen lati_main = atan(1)/45 * latitude_main

gen dlon=long_main-long_br
gen dlat=lati_main-lati_br
gen a=(sin(dlat/2))^2 + cos(lati_br) * cos(lati_main) * (sin(dlon/2))^2
gen c= 2 * atan2(sqrt(a),sqrt(1-a))
gen distance=3957 * c
/***use the other method to verify the result of distance, it's correct 
 gen distance2=2*3961*asin(sqrt(sin(dlat/2))^2)+cos(lati_br) * cos(lati_main)*(sin(dlon/2))^2
 drop distance2
 ***/
 codebook distance
 save 1999_dis_dep, replace

 /*************use latitude and longitude to calculate the distance between main office and branches****/
  clear
 cd "D:\Research\Banking power and IPO\dataneed"
 use  2010_dis_dep
gen long_br = atan(1)/45 * sims_latitude
gen lati_br = atan(1)/45 * sims_longitude
gen long_main = atan(1)/45 * latitude_main
gen lati_main = atan(1)/45 * latitude_main

gen dlon=long_main-long_br
gen dlat=lati_main-lati_br
gen a=(sin(dlat/2))^2 + cos(lati_br) * cos(lati_main) * (sin(dlon/2))^2
gen c= 2 * atan2(sqrt(a),sqrt(1-a))
gen distance=3957 * c
/***use the other method to verify the result of distance, it's correct 
 gen distance2=2*3961*asin(sqrt(sin(dlat/2))^2)+cos(lati_br) * cos(lati_main)*(sin(dlon/2))^2
 drop distance2
 ***/
 codebook distance
 save 2010_dis_dep, replace
